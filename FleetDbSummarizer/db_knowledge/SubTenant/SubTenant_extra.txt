Collection: SubTenant
Fields:
- primary_color: string (portal branding primary color)
- secondary_color: string (portal branding secondary color)
- created_by: string (user who created the record)
- logo: usually null (logo)
- logo_url: usually null (logo URL)
- updated_at: BSON Date (when the record was last changed)
- updated_by: usually null (user who last changed the record)
- email: string (account email)
- phone: string or null (account phone)
- container_id: int (container id)
- container_group_id: int (container group id)
- cr_number: string or null (commercial registration number)
- erp_id: string (ERP / billing id)
- account_manager: string (account manager of the customer)
- address: string (company address)
- billing_cycle: usually null (billing cycle of the account)
- divison: string (division (spelled 'divison' in the DB))
- order_booker: string (order booker of the customer)
- vat_number: string (company VAT number)
